def download_isd_year(year: int, max_retries: int = 3) -> "Path | None":
    gz_path  = OUTPUT_DIR / f"kord_{year}.gz"
    txt_path = OUTPUT_DIR / f"kord_{year}.txt"
    if txt_path.exists() and txt_path.stat().st_size > 10_000:
        print(f"  [Data] Cache hit: {txt_path}")
        return txt_path
    url = (f"https://www.ncei.noaa.gov/pub/data/noaa/isd-lite/"
           f"{year}/{STATION_USAF}-{STATION_WBAN}-{year}.gz")
    for attempt in range(1, max_retries + 1):
        try:
            print(f"  ⬇️  {year} (attempt {attempt})…")
            req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
            with urllib.request.urlopen(req, timeout=30) as r:
                with open(gz_path, "wb") as f:
                    f.write(r.read())
            break
        except Exception as e:
            print(f"   ⚠️  {e}")
            if attempt == max_retries:
                return None
    if not gz_path.exists():
        return None
    with gzip.open(gz_path, "rb") as f_in, open(txt_path, "wb") as f_out:
        shutil.copyfileobj(f_in, f_out)
    gz_path.unlink()
    return txt_path


def parse_isd_file(txt_path: Path) -> pd.DataFrame:
    rows = []
    with open(txt_path, "r", encoding="utf-8", errors="replace") as fh:
        for line in fh:
            parts = line.strip().split()
            if len(parts) < 10:
                continue
            try:
                yr, mo, dy, hr = int(parts[0]), int(parts[1]), int(parts[2]), int(parts[3])
                if not (1<=mo<=12 and 1<=dy<=31 and 0<=hr<=23): continue
                pf = lambda i: np.nan if (i>=len(parts) or parts[i]=="-9999") else int(parts[i])
                rows.append({"year":yr,"month":mo,"day":dy,"hour":hr,
                              "T_db":pf(4),"T_dew":pf(5),"SLP":pf(6),
                              "WD":pf(7),"WS":pf(8),"sky":pf(9)})
            except (ValueError, IndexError):
                continue
    df = pd.DataFrame(rows)
    df["timestamp"] = pd.to_datetime(df[["year","month","day","hour"]], errors="coerce")
    df = df.dropna(subset=["timestamp"]).set_index("timestamp").sort_index()
    for col in ["T_db","T_dew","SLP","WS"]:
        df[col] = df[col] / 10.0
    e_s = 6.1078 * np.exp(17.2694 * df["T_db"]  / (df["T_db"]  + 237.3))
    e_a = 6.1078 * np.exp(17.2694 * df["T_dew"] / (df["T_dew"] + 237.3))
    df["RH"] = (100.0 * e_a / e_s).clip(0, 100)
    return df


def load_isd_range(years=None) -> "pd.DataFrame | None":
    if years is None: years = DATA_YEARS
    frames = []
    for year in years:
        txt = download_isd_year(year)
        if txt is None:
            print(f"   ❌ {year} failed"); continue
        df_raw = parse_isd_file(txt)
        print(f"   📊 {year}: {len(df_raw):,} records")
        frames.append(df_raw)
    if not frames: return None
    df = pd.concat(frames).sort_index()
    full_idx = pd.date_range(df.index.min(), df.index.max(), freq="h")
    df = df.reindex(full_idx); df.index.name = "timestamp"
    for c in ["T_db","T_dew","SLP","WS","WD","RH"]:
        if c in df.columns: df[c] = df[c].interpolate(method="linear", limit=3)
    df = df.ffill().bfill()
    print(f"  ✅ {len(df):,} hourly slots  T_db=[{df.T_db.min():.1f},{df.T_db.max():.1f}]°C")
    return df[["T_db","T_dew","RH","WS","SLP","WD"]]


def make_synthetic_data(n=8760, seed=RNG_SEED):
    rng = np.random.default_rng(seed); t = np.arange(n)
    T_db  = 10*np.sin(2*np.pi*t/8760)+5*np.sin(2*np.pi*t/24)+rng.normal(0,1,n)
    T_dew = T_db - 5 + rng.normal(0,0.5,n)
    e_s   = 6.1078*np.exp(17.2694*T_db/(T_db+237.3))
    e_a   = 6.1078*np.exp(17.2694*T_dew/(T_dew+237.3))
    RH    = np.clip(100.0*e_a/e_s, 5, 100)
    WS    = np.abs(rng.normal(5,2,n))
    SLP   = 1013+5*np.sin(2*np.pi*t/8760)+rng.normal(0,1,n)
    WD    = rng.uniform(0,360,n)
    return pd.DataFrame({"T_db":T_db,"T_dew":T_dew,"RH":RH,"WS":WS,"SLP":SLP,"WD":WD})


def preprocess(df, window=WINDOW, horizons=None):
    """Chronological 75/15/10 split. FAST_MODE caps samples for quick testing."""
    if horizons is None: horizons = HORIZONS_FSDH
    scaler = StandardScaler()
    data   = scaler.fit_transform(df.values).astype(np.float32)
    X, y   = [], []
    max_h  = max(horizons)
    for i in range(window, len(data)-max_h):
        X.append(data[i-window:i])
        y.append([data[i+h-1, 0] for h in horizons])
    X = np.array(X, dtype=np.float32)
    y = np.array(y, dtype=np.float32)
    N = len(X)
    tr_end  = int(N*0.75); val_end = int(N*0.90)
    Xtr, ytr = X[:tr_end], y[:tr_end]
    Xva, yva = X[tr_end:val_end], y[tr_end:val_end]
    Xte, yte = X[val_end:], y[val_end:]
    if FAST_MODE:
        rng = np.random.default_rng(RNG_SEED)
        Xtr = Xtr[np.linspace(0,len(Xtr)-1,N_FAST_TRAIN,dtype=int)]
        ytr = ytr[np.linspace(0,len(ytr)-1,N_FAST_TRAIN,dtype=int)]
        Xva = Xva[np.linspace(0,len(Xva)-1,N_FAST_VAL,dtype=int)]
        yva = yva[np.linspace(0,len(yva)-1,N_FAST_VAL,dtype=int)]
        Xte = Xte[np.linspace(0,len(Xte)-1,N_FAST_TEST,dtype=int)]
        yte = yte[np.linspace(0,len(yte)-1,N_FAST_TEST,dtype=int)]
        print(f"  [Preprocess] FAST_MODE: capped to train={N_FAST_TRAIN} val={N_FAST_VAL} test={N_FAST_TEST}")
    sp = dict(X_train=Xtr,y_train=ytr,X_val=Xva,y_val=yva,X_test=Xte,y_test=yte,horizons=horizons)
    print(f"  [Preprocess] X_train={sp['X_train'].shape}, y_train={sp['y_train'].shape}")
    print(f"  [Preprocess] X_val={sp['X_val'].shape}, X_test={sp['X_test'].shape}")
    return sp, scaler

# ─────────────────────────────────────────────────────────────────────────────
# 2. ZZ FEATURE MAP ENCODER
# ─────────────────────────────────────────────────────────────────────────────

def zz_feature_map(x, n_qubits, depth=2):
    """
    ZZFeatureMap — Havlíček et al., Nature 567 (2019).
    H⊗n → Rz(2xi) → CNOT·Rz(2(π-xi)(π-xj))·CNOT  [×depth, all pairs]
    """
    for _ in range(depth):
        for i in range(n_qubits):
            qml.Hadamard(wires=i)
            qml.RZ(2.0*x[i], wires=i)
        for i, j in combinations(range(n_qubits), 2):
            qml.CNOT(wires=[i,j])
            qml.RZ(2.0*(np.pi-x[i])*(np.pi-x[j]), wires=j)
            qml.CNOT(wires=[i,j])

# ─────────────────────────────────────────────────────────────────────────────
# 3. TFIM RESERVOIR  (n_qubits-agnostic, projection scales with n_qubits)
# ─────────────────────────────────────────────────────────────────────────────

class TFIMReservoir:
    """
    H_TFIM = -J Σ ZiZj  - g Σ Xi   (periodic BC, J/g=1 critical point)
    Readout: ⟨Zi⟩ + ⟨ZiZi+1⟩  →  2·n_qubits features.

    FIX v3: projection matrix P has shape (24, n_qubits) — scales with
    n_qubits, so 20-qubit reservoir gets a full-rank projection, not
    zero-padded columns.
    """
    def __init__(self, n_qubits=N_QUBITS, trotter=TROTTER_STEPS,
                 j_over_g=J_OVER_G, encoder_depth=ENCODER_DEPTH,
                 backend="lightning.qubit"):
        self.n_qubits   = n_qubits
        self.trotter    = trotter
        self.enc_depth  = encoder_depth
        J = j_over_g/(j_over_g+1.0); g = 1.0-J
        self.J = J; self.g = g
        self.n_features = 2*n_qubits
        dt = 1.0/trotter

        dev = qml.device(backend, wires=n_qubits)

        @qml.qnode(dev, interface="numpy")
        def _stateless(x):
            zz_feature_map(x, n_qubits, depth=encoder_depth)
            for _ in range(trotter):
                for i in range(n_qubits): qml.IsingZZ(2*J*dt, wires=[i,(i+1)%n_qubits])
                for i in range(n_qubits): qml.RX(2*g*dt, wires=i)
            return ([qml.expval(qml.PauliZ(i)) for i in range(n_qubits)]
                   +[qml.expval(qml.PauliZ(i)@qml.PauliZ((i+1)%n_qubits))
                     for i in range(n_qubits)])

        @qml.qnode(dev, interface="numpy")
        def _sv_obs(sv, x):
            qml.StatePrep(sv, wires=range(n_qubits))
            zz_feature_map(x, n_qubits, depth=encoder_depth)
            for _ in range(trotter):
                for i in range(n_qubits): qml.IsingZZ(2*J*dt, wires=[i,(i+1)%n_qubits])
                for i in range(n_qubits): qml.RX(2*g*dt, wires=i)
            return ([qml.expval(qml.PauliZ(i)) for i in range(n_qubits)]
                   +[qml.expval(qml.PauliZ(i)@qml.PauliZ((i+1)%n_qubits))
                     for i in range(n_qubits)])

        @qml.qnode(dev, interface="numpy")
        def _sv_next(sv, x):
            qml.StatePrep(sv, wires=range(n_qubits))
            zz_feature_map(x, n_qubits, depth=encoder_depth)
            for _ in range(trotter):
                for i in range(n_qubits): qml.IsingZZ(2*J*dt, wires=[i,(i+1)%n_qubits])
                for i in range(n_qubits): qml.RX(2*g*dt, wires=i)
            return qml.state()

        self._stateless = _stateless
        self._sv_obs    = _sv_obs
        self._sv_next   = _sv_next

        # Projection: flatten last 4 timesteps × 6 features = 24 values → n_qubits
        # FIX: shape is always (24, n_qubits) regardless of qubit count
        rng = np.random.default_rng(0)
        P = rng.standard_normal((24, n_qubits))
        P /= np.linalg.norm(P, axis=0, keepdims=True) + 1e-12
        self._P = P

        print(f"  [Reservoir] J={J:.3f}  g={g:.3f}  J/g={J/g:.3f}  (critical @ 1.0)")
        print(f"  [Reservoir] {n_qubits} qubits · Trotter={trotter} · "
              f"ZZ-depth={encoder_depth} · backend={backend}")
        print(f"  [Reservoir] Readout dim: {self.n_features}  (⟨Z⟩ + ⟨ZZ⟩)")

    def encode(self, window: np.ndarray, noise_std: float = 0.0) -> np.ndarray:
        """
        (W,F) → x ∈ [0,π]^n_qubits via random projection of last 4 timesteps.
        noise_std > 0: input-encoding noise (correct way to regularise QRC —
        perturbs the quantum state before measurement, not the readout features).
        """
        x = window[-4:].flatten()[:24]
        if len(x) < 24: x = np.pad(x, (0, 24-len(x)))
        if noise_std > 0:
            x = x + np.random.default_rng().normal(0, noise_std, x.shape)
        proj = x @ self._P
        lo, hi = proj.min(), proj.max()
        return (proj-lo)/(hi-lo+1e-9) * np.pi

    def __call__(self, x):
        return np.array(self._stateless(x), dtype=np.float64)

    def transform(self, X, desc="", noise_std=0.0):
        N = len(X)
        out = np.zeros((N, self.n_features), dtype=np.float32)
        t0 = time.time()
        for i, w in enumerate(X):
            out[i] = self(self.encode(w, noise_std=noise_std))
            if (i+1) % max(1, N//4) == 0:
                el = time.time()-t0; eta = el/(i+1)*(N-i-1)
                print(f"    {desc} {i+1}/{N}  elapsed={el:.0f}s  eta={eta:.0f}s", end="\r")
        print()
        return out

# ─────────────────────────────────────────────────────────────────────────────
# 4. ESP VERIFICATION  (statevector-based, GS ⟹ ESP)
# ─────────────────────────────────────────────────────────────────────────────

def verify_esp(res, n_trials=ESP_TRIALS, seq_len=ESP_SEQ, seed=RNG_SEED):
    """
    Drives identical inputs from different computational-basis initial states.
    Pairwise observable distance should decay → 0 (contractive map ⟹ GS ⟹ ESP).
    Reference: Manjunath & Jaeger, Neural Computation 25(3), 2013.
    """
    print(f"\n  [ESP] Running {n_trials} trials × {seq_len} shared steps …")
    rng = np.random.default_rng(seed)
    dim = 2**res.n_qubits
    u   = rng.uniform(0, np.pi, (seq_len, res.n_qubits))
    init_svs = []
    for _ in range(n_trials):
        v = np.zeros(dim, dtype=complex); v[rng.integers(0,dim)] = 1.0
        init_svs.append(v)

    traj = []
    for ti, sv0 in enumerate(init_svs):
        sv = sv0.copy(); obs_seq = []
        for t in range(seq_len):
            obs = np.array(res._sv_obs(sv, u[t]), dtype=np.float64)
            sv  = np.array(res._sv_next(sv, u[t]), dtype=complex)
            obs_seq.append(obs)
        traj.append(obs_seq)
        print(f"    trial {ti+1}/{n_trials}", end="\r")
    print()

    traj  = np.array(traj)
    pairs = [(a,b) for a in range(n_trials) for b in range(a+1,n_trials)]
    dists = np.array([np.mean([np.linalg.norm(traj[a,t]-traj[b,t]) for a,b in pairs])
                      for t in range(seq_len)])

    ratio     = dists[-1]/(dists[0]+1e-12)
    converged = ratio < 0.20
    slope, _  = np.polyfit(np.arange(seq_len,dtype=float), np.log(dists+1e-12), 1)

    print(f"\n  [ESP] {'✅ ESP VERIFIED' if converged else '⚠️  NOT CONFIRMED'}")
    print(f"  [ESP] Initial divergence : {dists[0]:.4f}")
    print(f"  [ESP] Final   divergence : {dists[-1]:.4f}")
    print(f"  [ESP] Convergence ratio  : {ratio:.4f}  (< 0.20 ⟹ ESP)")
    print(f"  [ESP] Log-linear decay   : {slope:.5f}  "
          f"({'contractive ✓' if slope<0 else 'non-contractive ✗'})")
    print( "  [ESP] Proof: TFIM critical point ⟹ GS (contractivity) ⟹ ESP")
    return {"converged":converged,"ratio":float(ratio),"decay":float(slope),
            "curve":dists,"d0":float(dists[0]),"df":float(dists[-1])}

# ─────────────────────────────────────────────────────────────────────────────
# 5. MEMORY CAPACITY  (standard QRC benchmark)
# ─────────────────────────────────────────────────────────────────────────────

def compute_memory_capacity(res, X_val, y_val_col, max_lag=MC_MAX_LAG):
    """
    Memory Capacity (MC) — Jaeger 2001 / Dambre et al. 2012.

    MC = Σ_{k=1}^{L}  corr²( ŷ_k(t),  u(t-k) )

    where ŷ_k is the Ridge readout trained to recall the input k steps ago.
    MC > L/2 indicates the reservoir remembers more than half the past L inputs.
    At the critical point MC is near-maximal for the given qubit count.

    Uses Ridge (not KRR) for speed — this is a linear memory task.
    """
    print(f"  [MC] Extracting features for {len(X_val)} val samples…")
    Phi = res.transform(X_val, "MC-val")

    # Target: past values of T_db (feature 0) at each lag
    # y_val_col is the full T_db sequence aligned with X_val
    mc_total = 0.0
    mc_per_lag = []
    for lag in range(1, max_lag+1):
        if lag >= len(y_val_col): break
        target = y_val_col[:-lag] if lag > 0 else y_val_col
        feats  = Phi[lag:]
        if len(feats) < 20: break
        ridge = Ridge(alpha=1e-2)
        ridge.fit(feats, target)
        pred = ridge.predict(feats)
        r, _ = pearsonr(target, pred)
        mc_k  = r**2
        mc_total += mc_k
        mc_per_lag.append(mc_k)
        print(f"    lag={lag:2d}  corr²={mc_k:.4f}  MC_cumul={mc_total:.3f}", end="\r")
    print()
    print(f"  [MC] Total Memory Capacity = {mc_total:.4f}  "
          f"(max possible ≈ {max_lag}, target > {max_lag//2})")
    return {"mc": mc_total, "per_lag": mc_per_lag}

# ─────────────────────────────────────────────────────────────────────────────
# 6. KRR READOUT
# ─────────────────────────────────────────────────────────────────────────────

def train_krr(Xtr, ytr, Xval, yval):
    best_rmse, best = np.inf, None; bg, ba = None, None
    for g in KRR_GAMMAS:
        for a in KRR_ALPHAS:
            m = KernelRidge(kernel="rbf", gamma=g, alpha=a)
            m.fit(Xtr, ytr)
            r = np.sqrt(mean_squared_error(yval, m.predict(Xval)))
            if r < best_rmse: best_rmse, best, bg, ba = r, m, g, a
    print(f"    [KRR] gamma={bg}  alpha={ba:.2e}  val_RMSE={best_rmse:.4f}")
    return best

# 7. RESIDUAL QRC
# ─────────────────────────────────────────────────────────────────────────────

class ResidualQRC:
    """ŷ(t+h) = x(t) + KRR(Φ_QRC(x(t)))  where target = y(t+h) - persistence."""
    def __init__(self, res, noise_std=0.0):
        self.res = res; self.models = {}; self.scalers = {}
        self.noise_std = noise_std

    def _persist(self, X): return X[:,-1,0]

    def fit(self, sp, h_indices=None):
        horizons = sp["horizons"]
        if h_indices is not None: horizons = [horizons[i] for i in h_indices]
        Xtr, ytr = sp["X_train"], sp["y_train"]
        Xva, yva = sp["X_val"],   sp["y_val"]
        print(f"  [ResidualQRC] Reservoir features — train… (noise_std={self.noise_std})")
        Ptr = self.res.transform(Xtr, "train", noise_std=self.noise_std)
        print(f"  [ResidualQRC] Reservoir features — val…")
        Pva = self.res.transform(Xva, "val", noise_std=self.noise_std)
        for h in horizons:
            hi  = sp["horizons"].index(h)
            print(f"  [ResidualQRC] h={h}h residual KRR")
            rtr = ytr[:,hi] - self._persist(Xtr)
            rva = yva[:,hi] - self._persist(Xva)
            sc  = StandardScaler(); self.scalers[h] = sc
            self.models[h] = train_krr(Ptr, sc.fit_transform(rtr.reshape(-1,1)).ravel(),
                                        Pva, sc.transform(rva.reshape(-1,1)).ravel())

    def predict(self, X, horizons=None):
        if horizons is None: horizons = list(self.models.keys())
        P = self.res.transform(X, "test", noise_std=self.noise_std)
        return {h: self._persist(X) + self.scalers[h].inverse_transform(
                    self.models[h].predict(P).reshape(-1,1)).ravel()
                for h in horizons}

class DirectQRC:
    def __init__(self, res): self.res = res; self.models = {}

    def fit(self, sp):
        Xtr, ytr = sp["X_train"], sp["y_train"]
        Xva, yva = sp["X_val"],   sp["y_val"]
        print("  [DirectQRC] Reservoir features — train…")
        Ptr = self.res.transform(Xtr, "train")
        print("  [DirectQRC] Reservoir features — val…")
        Pva = self.res.transform(Xva, "val")
        for hi, h in enumerate(HORIZONS):
            print(f"  [DirectQRC] h={h}h KRR")
            self.models[h] = train_krr(Ptr, ytr[:,hi], Pva, yva[:,hi])

    def predict(self, X):
        P = self.res.transform(X, "test")
        return {h: self.models[h].predict(P) for h in HORIZONS}

def fsdh_curve(rqrc, sp, pers_r_all):
    """
    Forecast Skill Decay Horizon (FSDH) — correct definition:
    First horizon h at which the model's skill score DROPS BELOW zero
    (i.e. model stops beating persistence).

    FIX v3: was previously finding when persistence beats model (inverted).
    Now sweeps all horizons 1h..12h and returns the skill% at each.
    """
    horizons = sp["horizons"]
    Xt, yt   = sp["X_test"], sp["y_test"]
    skills   = {}
    print("  [FSDH] Predicting all horizons…")
    preds = rqrc.predict(Xt, horizons=horizons)
    for h in horizons:
        hi   = horizons.index(h)
        yt_h = yt[:,hi]
        rmse = np.sqrt(mean_squared_error(yt_h, preds[h]))
        sk   = 100*(1 - rmse/(pers_r_all[h]+1e-12))
        skills[h] = sk

    # FSDH = first h where skill < 0
    fsdh = next((h for h in horizons if skills[h] < 0), horizons[-1]+1)
    print(f"  [FSDH] Skill curve: " +
          "  ".join([f"h={h}h:{skills[h]:+.1f}%" for h in horizons]))
    print(f"  [FSDH] FSDH = {fsdh}h  (first horizon where skill < 0)")
    return skills, fsdh


def compute_ipc(Phi, y_col, n_delays=12):
    """IPC via mutual information — sum over lag delays."""
    from sklearn.feature_selection import mutual_info_regression
    scores = []
    for lag in range(1, min(n_delays, len(y_col))):
        y_lag = y_col[lag:]
        X_lag = Phi[:-lag]
        mi = mutual_info_regression(X_lag, y_lag, random_state=42)
        scores.append(np.sum(mi))
    return float(np.mean(scores)) if scores else 0.0


def noise_sweep(res, sp):
    """
    FIX v3: noise is applied to the *input encoding* (perturbs the quantum
    angles before circuit execution), not to the post-circuit readout features.
    This is the physically meaningful regularisation for QRC.
    """
    print("\n  [Noise] Extracting clean features once (baseline)…")
    horizons = HORIZONS
    results  = {}
    Xt, yt   = sp["X_test"], sp["y_test"]
    pers_p   = {h: Xt[:,-1,0] for h in horizons}

    for sigma in NOISE_LEVELS:
        print(f"\n  [Noise] sigma={sigma:.4f}  (input-encoding perturbation)")
        rqrc_n = ResidualQRC(res, noise_std=sigma)
        # Build a mini split using only HORIZONS (not all 12) to save time
        mini_sp = {k: v for k,v in sp.items()}
        mini_sp["horizons"] = horizons
        # Re-index y to only HORIZONS columns
        h_idx = [sp["horizons"].index(h) for h in horizons]
        for split in ["train","val","test"]:
            mini_sp[f"y_{split}"] = sp[f"y_{split}"][:, h_idx]
        rqrc_n.fit(mini_sp)
        preds  = rqrc_n.predict(Xt, horizons=horizons)
        res_rmse = {}
        for h in horizons:
            hi      = horizons.index(h)
            rmse    = np.sqrt(mean_squared_error(yt[:,h_idx[hi]], preds[h]))
            res_rmse[h] = rmse
            print(f"    h={h}h  test_RMSE={rmse:.4f}")
        results[sigma] = res_rmse
    return results
