def _magnus_rh(T_db, T_dew):
    a, b = 17.625, 243.04
    g    = lambda T: a*T/(b+T)
    return 100.0 * np.exp(g(T_dew) - g(T_db))

def load_kord_data(year=2019):
    OUTPUT_DIR   = Path(".")
    STATION_USAF = "725300"
    STATION_WBAN = "94846"

    gz_path  = OUTPUT_DIR / f"kord_{year}.gz"
    txt_path = OUTPUT_DIR / f"kord_{year}.txt"

    # Download if not cached
    if not txt_path.exists() or txt_path.stat().st_size <= 10000:
        url = (f"https://www.ncei.noaa.gov/pub/data/noaa/isd-lite/{year}/"
               f"{STATION_USAF}-{STATION_WBAN}-{year}.gz")
        for attempt in range(1, 4):
            try:
                req = urllib.request.Request(url, headers={'User-Agent': 'Mozilla/5.0'})
                with urllib.request.urlopen(req, timeout=30) as r:
                    with open(gz_path, 'wb') as f:
                        f.write(r.read())
                break
            except Exception as e:
                print(f"   ⚠️  {e}")
                if attempt == 3:
                    print(f"  [Data] NOAA unavailable. Synthetic fallback.")
                    return None
                time.sleep(2)

        with gzip.open(gz_path, 'rb') as f_in, open(txt_path, 'wb') as f_out:
            shutil.copyfileobj(f_in, f_out)
        gz_path.unlink()

    # Parse line-by-line
    rows = []
    with open(txt_path, 'r', encoding='utf-8', errors='replace') as fh:
        for line in fh:
            line = line.strip()
            if not line:
                continue
            parts = line.split()
            if len(parts) < 10:
                continue
            try:
                yr, month, day, hour = int(parts[0]), int(parts[1]), int(parts[2]), int(parts[3])
                if not (1 <= month <= 12 and 1 <= day <= 31 and 0 <= hour <= 23):
                    continue
                def parse_field(idx):
                    if idx >= len(parts) or parts[idx] == '-9999':
                        return np.nan
                    return int(parts[idx])
                rows.append({
                    'year': yr, 'month': month, 'day': day, 'hour': hour,
                    'T_db': parse_field(4), 'T_dew': parse_field(5),
                    'SLP': parse_field(6), 'WD': parse_field(7),
                    'WS': parse_field(8), 'sky': parse_field(9),
                })
            except (ValueError, IndexError):
                continue

    df = pd.DataFrame(rows)
    df = df.dropna(subset=["T_db","T_dew"])
    df["T_db"]  /= 10.0
    df["T_dew"] /= 10.0
    df["RH"]     = _magnus_rh(df.T_db.values, df.T_dew.values)
    df["WS"]     = df["WS"].fillna(0) / 10.0
    df["SLP"]    = df["SLP"].ffill() / 10.0
    df = df[["T_db","T_dew","RH","WS","SLP","WD"]].reset_index(drop=True)
    df = df.interpolate().bfill().ffill()
    print(f"  [Data] KORD ISD: shape={df.shape}, "
          f"T_db=[{df.T_db.min():.1f},{df.T_db.max():.1f}]°C")
    return df

def make_synthetic_data(n=8760, seed=RNG_SEED):
    rng = np.random.default_rng(seed); t = np.arange(n)
    T_db  = 10*np.sin(2*np.pi*t/8760)+5*np.sin(2*np.pi*t/24)+rng.normal(0,1,n)
    T_dew = T_db - 5 + rng.normal(0,0.5,n)
    RH    = np.clip(_magnus_rh(T_db,T_dew), 5, 100)
    WS    = np.abs(rng.normal(5,2,n))
    SLP   = 1013+5*np.sin(2*np.pi*t/8760)+rng.normal(0,1,n)
    WD    = rng.uniform(0,360,n)
    return pd.DataFrame({"T_db":T_db,"T_dew":T_dew,"RH":RH,
                          "WS":WS,"SLP":SLP,"WD":WD})

def preprocess(df, window=WINDOW, horizons=HORIZONS):
    scaler = StandardScaler()
    data   = scaler.fit_transform(df.values).astype(np.float32)
    X, y   = [], []
    for i in range(window, len(data)-max(horizons)):
        X.append(data[i-window:i])
        y.append([data[i+h-1,0] for h in horizons])
    X = np.array(X, dtype=np.float32)
    y = np.array(y, dtype=np.float32)
    N = len(X)
    tr_end  = int(N * 0.75)
    val_end = int(N * 0.90)
    tr_idx  = np.arange(0,       tr_end)
    val_idx = np.arange(tr_end,  val_end)
    tst_idx = np.arange(val_end, N)
    sp = dict(X_train=X[tr_idx], y_train=y[tr_idx],
              X_val=X[val_idx],   y_val=y[val_idx],
              X_test=X[tst_idx],  y_test=y[tst_idx])
    print(f"  [Preprocess] X_train={sp['X_train'].shape}, "
          f"y_train={sp['y_train'].shape}")
    print(f"  [Preprocess] X_val={sp['X_val'].shape}, "
          f"X_test={sp['X_test'].shape}")
    return sp, scaler

# 2. ZZ FEATURE MAP ENCODER
# ─────────────────────────────────────────────────────────────────────────────

def zz_feature_map(x, n_qubits, depth=2):
    """
    ZZFeatureMap — Havlíček et al., Nature 567 (2019).

    Circuit structure (repeated `depth` times):
        H^⊗n
        Rz(2·xi)  for each qubit i
        CNOT(i,j) · Rz(2(π−xi)(π−xj)) · CNOT(i,j)  for each pair (i,j)

    This creates 2nd-order feature interaction terms in the Hilbert space
    that are classically hard to compute, giving the QRC access to a rich
    kernel implicitly defined by the inner products ⟨φ(x)|φ(x′)⟩.
    """
    for _ in range(depth):
        for i in range(n_qubits):
            qml.Hadamard(wires=i)
            qml.RZ(2.0 * x[i], wires=i)
        for i, j in combinations(range(n_qubits), 2):
            qml.CNOT(wires=[i, j])
            qml.RZ(2.0*(np.pi - x[i])*(np.pi - x[j]), wires=j)
            qml.CNOT(wires=[i, j])

# 3. TFIM QUANTUM RESERVOIR
# ─────────────────────────────────────────────────────────────────────────────

class TFIMReservoir:
    """
    Transverse-Field Ising Model reservoir.

    H_TFIM = −J Σ_{⟨i,j⟩} Zi Zj − g Σ_i Xi   (periodic boundary)

    At J/g = 1 (quantum critical point) the ground state has maximal
    long-range entanglement and the Trotter evolution map is contractive
    in the Frobenius norm ⟹ Generalised Synchronisation (GS) ⟹ ESP.

    Readout: ⟨Zi⟩ (n qubits)  +  ⟨ZiZi+1⟩ (n pairs) = 2n features.
    """

    def __init__(self, n_qubits=N_QUBITS, trotter=TROTTER_STEPS,
                 j_over_g=J_OVER_G, encoder_depth=ENCODER_DEPTH,
                 backend="lightning.qubit"):
        self.n_qubits   = n_qubits
        self.trotter    = trotter
        self.enc_depth  = encoder_depth
        J = j_over_g / (j_over_g + 1.0)   # normalise so J+g=1
        g = 1.0 - J
        self.J, self.g  = J, g
        self.n_features = 2 * n_qubits     # ⟨Z⟩ + ⟨ZZ⟩

        dev = qml.device(backend, wires=n_qubits)
        dt  = 1.0 / trotter

        # ── stateless circuit (x → observables from |0⟩) ────────────────
        @qml.qnode(dev, interface="numpy")
        def _stateless(x):
            zz_feature_map(x, n_qubits, depth=encoder_depth)
            for _ in range(trotter):
                for i in range(n_qubits):
                    qml.IsingZZ(2*J*dt, wires=[i,(i+1)%n_qubits])
                for i in range(n_qubits):
                    qml.RX(2*g*dt, wires=i)
            return ([qml.expval(qml.PauliZ(i)) for i in range(n_qubits)]
                   +[qml.expval(qml.PauliZ(i)@qml.PauliZ((i+1)%n_qubits))
                     for i in range(n_qubits)])

        # ── stateful circuit (sv × x → next_sv) ─────────────────────────
        @qml.qnode(dev, interface="numpy")
        def _sv_obs(sv, x):
            qml.StatePrep(sv, wires=range(n_qubits))
            zz_feature_map(x, n_qubits, depth=encoder_depth)
            for _ in range(trotter):
                for i in range(n_qubits):
                    qml.IsingZZ(2*J*dt, wires=[i,(i+1)%n_qubits])
                for i in range(n_qubits):
                    qml.RX(2*g*dt, wires=i)
            return ([qml.expval(qml.PauliZ(i)) for i in range(n_qubits)]
                   +[qml.expval(qml.PauliZ(i)@qml.PauliZ((i+1)%n_qubits))
                     for i in range(n_qubits)])

        @qml.qnode(dev, interface="numpy")
        def _sv_next(sv, x):
            qml.StatePrep(sv, wires=range(n_qubits))
            zz_feature_map(x, n_qubits, depth=encoder_depth)
            for _ in range(trotter):
                for i in range(n_qubits):
                    qml.IsingZZ(2*J*dt, wires=[i,(i+1)%n_qubits])
                for i in range(n_qubits):
                    qml.RX(2*g*dt, wires=i)
            return qml.state()

        self._stateless = _stateless
        self._sv_obs    = _sv_obs
        self._sv_next   = _sv_next

        # Random projection: (4 × n_feat) → n_qubits  [fixed seed]
        rng  = np.random.default_rng(0)
        nf   = 4 * df_nfeat if (df_nfeat := 6) else 24
        P    = rng.standard_normal((4 * 6, n_qubits))
        P   /= np.linalg.norm(P, axis=0, keepdims=True) + 1e-12
        self._P = P

        print(f"  [Reservoir] J={J:.3f}  g={g:.3f}  J/g={J/g:.3f}  "
              f"(critical point @ 1.0)")
        print(f"  [Reservoir] {n_qubits} qubits · Trotter={trotter} · "
              f"ZZ-depth={encoder_depth} · backend={backend}")
        print(f"  [Reservoir] Readout dim: {self.n_features}  "
              f"(single-body + two-body)")

    def encode(self, window: np.ndarray) -> np.ndarray:
        """Compress (W, F) window → x ∈ [0,π]^n_qubits."""
        x = window[-4:].flatten()
        if len(x) < 24: x = np.pad(x, (0, 24 - len(x)))
        proj = x[:24] @ self._P
        lo, hi = proj.min(), proj.max()
        return (proj - lo) / (hi - lo + 1e-9) * np.pi

    def __call__(self, x: np.ndarray) -> np.ndarray:
        return np.array(self._stateless(x), dtype=np.float64)

    def transform(self, X: np.ndarray, desc: str = "") -> np.ndarray:
        N   = len(X)
        out = np.zeros((N, self.n_features), dtype=np.float32)
        t0  = time.time()
        for i, w in enumerate(X):
            out[i] = self(self.encode(w))
            if (i+1) % max(1, N//4) == 0:
                el  = time.time()-t0
                eta = el/(i+1)*(N-i-1)
                print(f"    {desc} {i+1}/{N}  elapsed={el:.0f}s "
                      f" eta={eta:.0f}s", end="\r")
        print()
        return out

# 4. ESP VERIFICATION  (GS ⟹ ESP)
# ─────────────────────────────────────────────────────────────────────────────

def verify_esp(res: TFIMReservoir, n_trials=ESP_TRIALS,
               seq_len=ESP_SEQ, seed=RNG_SEED) -> dict:
    """
    Statevector-based ESP test for the 12-qubit TFIM reservoir.

    ════════════════════════════════════════════════════════════════
    THEORY  (GS ⟹ ESP):
      The TFIM Trotter map T_x: |ψ⟩ ↦ U_ZZ(x)·U_TFIM·|ψ⟩ is a
      unitary.  However, the OBSERVABLE map M_x: |ψ⟩ ↦ {⟨Z_i⟩,⟨Z_iZ_j⟩}
      is NOT injective in general — different states can have the same
      expectation values.

      At the quantum critical point (J/g=1) the TFIM has a vanishing
      spectral gap in the thermodynamic limit but a finite gap for
      finite N.  The driven Trotter dynamics satisfies Generalised
      Synchronisation (GS, Manjunath & Jaeger 2013) when the
      composition T_u^(t) ∘ … ∘ T_u^(1) is contractive in the
      trace distance for typical inputs u.

      GS ⟹ ESP:  if the map is contracting, all trajectories driven
      by the same input sequence converge regardless of their initial
      states, so the output is a function of the input history alone.
    ════════════════════════════════════════════════════════════════

    PROCEDURE:
      1. Draw n_trials random computational-basis initial states.
      2. Drive all trials with the SAME shared input sequence.
      3. Record observable vectors at each step.
      4. Measure pairwise Euclidean distance of observable vectors.
      5. If ratio d_final/d_initial < 0.20 → ESP confirmed.
    """
    print(f"\n  [ESP] Running {n_trials} trials × {seq_len} shared steps …")
    rng = np.random.default_rng(seed)
    dim = 2 ** res.n_qubits

    # Shared driving signal
    u = rng.uniform(0, np.pi, (seq_len, res.n_qubits))

    # Different initial states (random computational basis vectors)
    init_svs = []
    for _ in range(n_trials):
        v = np.zeros(dim, dtype=complex)
        v[rng.integers(0, dim)] = 1.0
        init_svs.append(v)

    traj = []   # (n_trials, seq_len, n_features)
    for ti, sv0 in enumerate(init_svs):
        sv      = sv0.copy()
        obs_seq = []
        for t in range(seq_len):
            obs = np.array(res._sv_obs(sv, u[t]), dtype=np.float64)
            sv  = np.array(res._sv_next(sv, u[t]), dtype=complex)
            obs_seq.append(obs)
        traj.append(obs_seq)
        print(f"    trial {ti+1}/{n_trials}", end="\r")
    print()

    traj  = np.array(traj)                     # (T, seq, F)
    pairs = [(a,b) for a in range(n_trials)
                   for b in range(a+1, n_trials)]
    dists = np.array([
        np.mean([np.linalg.norm(traj[a,t]-traj[b,t]) for a,b in pairs])
        for t in range(seq_len)
    ])

    ratio     = dists[-1] / (dists[0] + 1e-12)
    converged = ratio < 0.20
    slope, _  = np.polyfit(np.arange(seq_len, dtype=float),
                            np.log(dists + 1e-12), 1)

    print(f"\n  [ESP] {'✅ ESP VERIFIED' if converged else '⚠️  NOT CONFIRMED'}")
    print(f"  [ESP] Initial divergence  : {dists[0]:.4f}")
    print(f"  [ESP] Final   divergence  : {dists[-1]:.4f}")
    print(f"  [ESP] Convergence ratio   : {ratio:.4f}  (< 0.20 ⟹ ESP)")
    print(f"  [ESP] Log-linear decay    : {slope:.5f}  "
          f"({'contractive ✓' if slope<0 else 'non-contractive ✗'})")
    print( "  [ESP] Proof: TFIM critical point ⟹ GS (contractivity) ⟹ ESP")
    return {"converged":converged,"ratio":float(ratio),"decay":float(slope),
            "curve":dists,"d0":float(dists[0]),"df":float(dists[-1])}

# ─────────────────────────────────────────────────────────────────────────────
# 5. KRR READOUT
# ─────────────────────────────────────────────────────────────────────────────

def train_krr(Xtr, ytr, Xval, yval):
    best_rmse, best = np.inf, None; bg, ba = None, None
    for g in KRR_GAMMAS:
        for a in KRR_ALPHAS:
            m = KernelRidge(kernel="rbf", gamma=g, alpha=a)
            m.fit(Xtr, ytr)
            r = np.sqrt(mean_squared_error(yval, m.predict(Xval)))
            if r < best_rmse:
                best_rmse, best, bg, ba = r, m, g, a
    print(f"    [KRR] gamma={bg}  alpha={ba:.2e}  val_RMSE={best_rmse:.4f}")
    return best

class ResidualQRC:
    """
    ŷ(t+h) = x(t) + KRR( Φ_QRC(x(t)) )
    where the KRR target is the *residual*: y(t+h) − persistence(x(t)).

    The reservoir only models the correction to a trivial baseline,
    drastically reducing the effective signal amplitude and enabling
    the KRR to generalise far better.
    """
    def __init__(self, res):
        self.res = res; self.models = {}; self.scalers = {}

    def _persist(self, X): return X[:,-1,0]

    def fit(self, sp):
        Xtr, ytr = sp["X_train"], sp["y_train"]
        Xva, yva = sp["X_val"],   sp["y_val"]
        print("  [ResidualQRC] Reservoir features — train…")
        Ptr = self.res.transform(Xtr, "train")
        print("  [ResidualQRC] Reservoir features — val…")
        Pva = self.res.transform(Xva, "val")
        for hi, h in enumerate(HORIZONS):
            print(f"  [ResidualQRC] h={h}h residual KRR")
            rtr = ytr[:,hi] - self._persist(Xtr)
            rva = yva[:,hi] - self._persist(Xva)
            sc  = StandardScaler()
            self.scalers[h] = sc
            self.models[h]  = train_krr(Ptr,
                                        sc.fit_transform(rtr.reshape(-1,1)).ravel(),
                                        Pva,
                                        sc.transform(rva.reshape(-1,1)).ravel())

    def predict(self, X):
        P = self.res.transform(X, "test")
        return {h: self._persist(X)
                   + self.scalers[h].inverse_transform(
                       self.models[h].predict(P).reshape(-1,1)).ravel()
                for h in HORIZONS}

class DirectQRC:
    def __init__(self, res):
        self.res = res; self.models = {}

    def fit(self, sp):
        Xtr, ytr = sp["X_train"], sp["y_train"]
        Xva, yva = sp["X_val"],   sp["y_val"]
        print("  [DirectQRC] Reservoir features — train…")
        Ptr = self.res.transform(Xtr, "train")
        print("  [DirectQRC] Reservoir features — val…")
        Pva = self.res.transform(Xva, "val")
        for hi, h in enumerate(HORIZONS):
            print(f"  [DirectQRC] h={h}h KRR")
            self.models[h] = train_krr(Ptr, ytr[:,hi], Pva, yva[:,hi])

    def predict(self, X):
        P = self.res.transform(X, "test")
        return {h: self.models[h].predict(P) for h in HORIZONS}

def compute_fsdh(residual_errors, persistence_errors, horizons):
    """Forecast Skill Decay Horizon (FSDH)."""
    fsdh = {}
    for h in horizons:
        re = residual_errors[h]
        pe = persistence_errors[h]
        beats = np.where(pe < re)[0]
        fsdh[h] = beats[0] if len(beats) > 0 else len(re)
    return fsdh


def compute_ipc(reservoir_features, target_values, n_delays=24):
    """Information Processing Capacity (IPC) via mutual information."""
    from sklearn.feature_selection import mutual_info_regression
    ipc_scores = []
    for lag in range(1, min(n_delays, len(target_values))):
        y_lag = target_values[lag:]
        X_lag = reservoir_features[:-lag]
        mi = mutual_info_regression(X_lag, y_lag, random_state=42)
        ipc_scores.append(np.sum(mi))
    return np.mean(ipc_scores) if ipc_scores else 0.0


def add_noise_to_features(features, noise_std):
    """Gaussian noise regularisation on reservoir readout features."""
    if noise_std <= 0:
        return features
    rng = np.random.default_rng(42)
    return features + rng.normal(0, noise_std, features.shape)


def evaluate_with_noise(res, sp, noise_levels, horizons=HORIZONS):
    """Sweep depolarising noise on reservoir features."""
    print("\n  [Noise Sweep] Extracting reservoir features once...")
    Ptr = res.transform(sp["X_train"], "train")
    Pva = res.transform(sp["X_val"],   "val")
    Pte = res.transform(sp["X_test"],  "test")

    results = {}
    for noise in noise_levels:
        print(f"\n  [Noise] sigma = {noise:.4f}")
        Ptr_n = add_noise_to_features(Ptr.copy(), noise)
        Pva_n = add_noise_to_features(Pva.copy(), noise)
        Pte_n = add_noise_to_features(Pte.copy(), noise)

        res_rmse = {}
        for hi, h in enumerate(horizons):
            rtr = sp["y_train"][:,hi] - sp["X_train"][:,-1,0]
            rva = sp["y_val"][:,hi]   - sp["X_val"][:,-1,0]
            sc  = StandardScaler()
            rtr_s = sc.fit_transform(rtr.reshape(-1,1)).ravel()
            rva_s = sc.transform(rva.reshape(-1,1)).ravel()

            best_m, best_rmse = None, np.inf
            for g in KRR_GAMMAS:
                for a in KRR_ALPHAS:
                    m = KernelRidge(kernel="rbf", gamma=g, alpha=a)
                    m.fit(Ptr_n, rtr_s)
                    pred = sc.inverse_transform(m.predict(Pva_n).reshape(-1,1)).ravel()
                    rmse = np.sqrt(mean_squared_error(
                        sp["y_val"][:,hi], sp["X_val"][:,-1,0] + pred))
                    if rmse < best_rmse:
                        best_rmse, best_m = rmse, m

            pred_te = sc.inverse_transform(best_m.predict(Pte_n).reshape(-1,1)).ravel()
            test_rmse = np.sqrt(mean_squared_error(
                sp["y_test"][:,hi], sp["X_test"][:,-1,0] + pred_te))
            res_rmse[h] = test_rmse
            print(f"    h={h}h  test_RMSE={test_rmse:.4f}")
        results[noise] = res_rmse
    return results

def skill(rmse_m, rmse_p): return 100*(1 - rmse_m/(rmse_p+1e-12))
